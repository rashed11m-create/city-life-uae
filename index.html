function makePerson(clothesColor, skinColor = 0xc58b60) { const person = new THREE.Group(); const clothes = material(clothesColor); const skin = material(skinColor); const dark = material(0x252525); // الجسم const torso = new THREE.Mesh( new THREE.CapsuleGeometry(0.29, 0.48, 5, 10), clothes ); torso.position.y = 1.25; person.add(torso); // الرأس const head = new THREE.Mesh( new THREE.SphereGeometry(0.22, 20, 16), skin ); head.position.y = 1.95; person.add(head); // الشعر const hair = new THREE.Mesh( new THREE.SphereGeometry( 0.225, 16, 10, 0, Math.PI * 2, 0, Math.PI * 0.48 ), dark ); hair.position.y = 2.04; person.add(hair); // الذراعان function createArm(x) { const pivot = new THREE.Group(); pivot.position.set(x, 1.58, 0); person.add(pivot); const arm = new THREE.Mesh( new THREE.CapsuleGeometry(0.085, 0.38, 4, 8), clothes ); arm.position.y = -0.25; pivot.add(arm); const hand = new THREE.Mesh( new THREE.SphereGeometry(0.085, 12, 10), skin ); hand.position.y = -0.53; pivot.add(hand); return pivot; } const leftArm = createArm(-0.36); const rightArm = createArm(0.36); // الرجلان والقدمان function createLeg(x) { const pivot = new THREE.Group(); pivot.position.set(x, 0.83, 0); person.add(pivot); const leg = new THREE.Mesh( new THREE.CapsuleGeometry(0.105, 0.43, 5, 8), dark ); leg.position.y = -0.28; pivot.add(leg); const foot = new THREE.Mesh( new THREE.BoxGeometry(0.22, 0.12, 0.34), dark ); foot.position.set(0, -0.55, 0.09); pivot.add(foot); return pivot; } const leftLeg = createLeg(-0.17); const rightLeg = createLeg(0.17); person.userData = { leftArm, rightArm, leftLeg, rightLeg, phase: 0, body: torso, head }; return person; } function animatePerson(person, dt, amount, run = false) { const p = person.userData; const intensity = THREE.MathUtils.clamp(amount, 0, 1); const moving = intensity > 0.035; if (moving) { p.phase += dt * (run ? 13 : 8); } const swing = moving ? Math.sin(p.phase) * (run ? 0.65 : 0.42) * intensity : 0; // حركة الرجلين p.leftLeg.rotation.x = THREE.MathUtils.damp( p.leftLeg.rotation.x, swing, 12, dt ); p.rightLeg.rotation.x = THREE.MathUtils.damp( p.rightLeg.rotation.x, -swing, 12, dt ); // حركة اليدين بعكس الرجلين p.leftArm.rotation.x = THREE.MathUtils.damp( p.leftArm.rotation.x, -swing * 0.8, 12, dt ); p.rightArm.rotation.x = THREE.MathUtils.damp( p.rightArm.rotation.x, swing * 0.8, 12, dt ); // حركة الجسم const bounce = moving ? Math.abs(Math.sin(p.phase * 2)) * (run ? 0.055 : 0.025) * intensity : 0; person.position.y = THREE.MathUtils.damp( person.position.y, bounce, 12, dt ); // ميل الجسم أثناء الركض p.body.rotation.x = THREE.MathUtils.damp( p.body.rotation.x, run && moving ? -0.09 : 0, 8, dt ); // حركة الرأس p.head.rotation.x = THREE.MathUtils.damp( p.head.rotation.x, moving && run ? -0.04 : 0, 8, dt ); }
